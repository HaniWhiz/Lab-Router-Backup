#!/bin/sh

PASSWORD='L4bBackup!2026'
BACKUP_DIR='/root/cisco-backups'

mkdir -p "$BACKUP_DIR"

for i in 1 2 3 4
do
    sshpass -p "$PASSWORD" scp -O -6 \
        -o KexAlgorithms=+diffie-hellman-group14-sha1 \
        -o HostKeyAlgorithms=+ssh-rsa \
        -o StrictHostKeyChecking=no \
        -o UserKnownHostsFile=/dev/null \
        "backup@[2001:db8:1:ffff::$i]:running-config" \
        "$BACKUP_DIR/R$i-config.cfg"
done
EOF

chmod 700 /root/cisco-backup.sh
